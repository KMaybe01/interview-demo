## 1. 工作流引擎实现

### 1.1 工作流节点定义

```go
// internal/workflow/node.go
package workflow

import (
    "context"
    "fmt"
)

// NodeType 节点类型
type NodeType string

const (
    NodeTypeStart     NodeType = "start"
    NodeTypeEnd       NodeType = "end"
    NodeTypeLLM       NodeType = "llm"
    NodeTypeCode      NodeType = "code"
    NodeTypeTool      NodeType = "tool"
    NodeTypeCondition NodeType = "condition"
)

// Node 工作流节点接口
type Node interface {
    Execute(ctx context.Context, state *State) (*State, error)
    GetType() NodeType
    GetID() string
}

// BaseNode 基础节点
type BaseNode struct {
    ID   string
    Type NodeType
}

func (n *BaseNode) GetID() string {
    return n.ID
}

func (n *BaseNode) GetType() NodeType {
    return n.Type
}

// LLMNode LLM 节点
type LLMNode struct {
    BaseNode
    Prompt    string
    Model     string
    LLMClient LLMClient
}

type LLMClient interface {
    Chat(ctx context.Context, prompt string) (string, error)
}

func NewLLMNode(id string, prompt string, model string, client LLMClient) *LLMNode {
    return &LLMNode{
        BaseNode: BaseNode{ID: id, Type: NodeTypeLLM},
        Prompt:   prompt,
        Model:    model,
        LLMClient: client,
    }
}

func (n *LLMNode) Execute(ctx context.Context, state *State) (*State, error) {
    // 替换 prompt 中的变量
    prompt := n.Prompt
    for key, value := range state.Variables {
        placeholder := fmt.Sprintf("{{%s}}", key)
        prompt = strings.ReplaceAll(prompt, placeholder, fmt.Sprintf("%v", value))
    }

    // 调用 LLM
    response, err := n.LLMClient.Chat(ctx, prompt)
    if err != nil {
        return nil, fmt.Errorf("LLM 调用失败: %w", err)
    }

    // 更新状态
    newState := state.Clone()
    newState.Variables["llm_output"] = response
    newState.CurrentNode = n.ID

    return newState, nil
}

// CodeNode 代码节点
type CodeNode struct {
    BaseNode
    Code string
}

func NewCodeNode(id string, code string) *CodeNode {
    return &CodeNode{
        BaseNode: BaseNode{ID: id, Type: NodeTypeCode},
        Code:     code,
    }
}

func (n *CodeNode) Execute(ctx context.Context, state *State) (*State, error) {
    // 简单的代码执行（实际应该使用沙箱）
    newState := state.Clone()
    newState.Variables["code_result"] = "代码执行结果"
    newState.CurrentNode = n.ID

    return newState, nil
}

// ConditionNode 条件节点
type ConditionNode struct {
    BaseNode
    Condition string
    TrueNode  string
    FalseNode string
}

func NewConditionNode(id, condition, trueNode, falseNode string) *ConditionNode {
    return &ConditionNode{
        BaseNode:  BaseNode{ID: id, Type: NodeTypeCondition},
        Condition: condition,
        TrueNode:  trueNode,
        FalseNode: falseNode,
    }
}

func (n *ConditionNode) Execute(ctx context.Context, state *State) (*State, error) {
    newState := state.Clone()
    newState.CurrentNode = n.ID

    // 简单的条件判断
    if result, ok := state.Variables[n.Condition]; ok {
        if boolResult, ok := result.(bool); ok && boolResult {
            newState.NextNode = n.TrueNode
        } else {
            newState.NextNode = n.FalseNode
        }
    } else {
        newState.NextNode = n.FalseNode
    }

    return newState, nil
}
```

### 1.2 工作流状态

```go
// internal/workflow/state.go
package workflow

import "github.com/google/uuid"

// State 工作流状态
type State struct {
    ID          string
    Input       string
    Output      string
    CurrentNode string
    NextNode    string
    Variables   map[string]interface{}
    History     []NodeExecution
}

type NodeExecution struct {
    NodeID    string
    Input     string
    Output    string
    Duration  int64
    Error     string
}

func NewState(input string) *State {
    return &State{
        ID:        uuid.New().String(),
        Input:     input,
        Variables: make(map[string]interface{}),
        History:   []NodeExecution{},
    }
}

func (s *State) Clone() *State {
    newVars := make(map[string]interface{})
    for k, v := range s.Variables {
        newVars[k] = v
    }

    newHistory := make([]NodeExecution, len(s.History))
    copy(newHistory, s.History)

    return &State{
        ID:          s.ID,
        Input:       s.Input,
        Output:      s.Output,
        CurrentNode: s.CurrentNode,
        NextNode:    s.NextNode,
        Variables:   newVars,
        History:     newHistory,
    }
}

func (s *State) AddHistory(execution NodeExecution) {
    s.History = append(s.History, execution)
}
```

### 1.3 工作流引擎

```go
// internal/workflow/engine.go
package workflow

import (
    "context"
    "fmt"
    "sync"
    "time"
)

// Edge 工作流边
type Edge struct {
    Source    string
    Target    string
    Condition func(state *State) bool
}

// Engine 工作流引擎
type Engine struct {
    Nodes map[string]Node
    Edges []Edge
    mu    sync.RWMutex
}

func NewEngine() *Engine {
    return &Engine{
        Nodes: make(map[string]Node),
        Edges: []Edge{},
    }
}

func (e *Engine) AddNode(node Node) {
    e.mu.Lock()
    defer e.mu.Unlock()
    e.Nodes[node.GetID()] = node
}

func (e *Engine) AddEdge(source, target string, condition func(state *State) bool) {
    e.mu.Lock()
    defer e.mu.Unlock()
    e.Edges = append(e.Edges, Edge{
        Source:    source,
        Target:    target,
        Condition: condition,
    })
}

func (e *Engine) Execute(ctx context.Context, input string) (*State, error) {
    state := NewState(input)
    currentNodeID := "start"

    for {
        // 检查是否到达结束节点
        if currentNodeID == "end" {
            break
        }

        // 获取当前节点
        node, exists := e.Nodes[currentNodeID]
        if !exists {
            return nil, fmt.Errorf("节点 %s 不存在", currentNodeID)
        }

        // 执行节点
        start := time.Now()
        newState, err := node.Execute(ctx, state)
        duration := time.Since(start).Milliseconds()

        if err != nil {
            state.AddHistory(NodeExecution{
                NodeID:   currentNodeID,
                Input:    state.Input,
                Duration: duration,
                Error:    err.Error(),
            })
            return nil, fmt.Errorf("节点 %s 执行失败: %w", currentNodeID, err)
        }

        // 记录执行历史
        state.AddHistory(NodeExecution{
            NodeID:   currentNodeID,
            Input:    state.Input,
            Output:   newState.Output,
            Duration: duration,
        })

        // 更新状态
        state = newState

        // 查找下一个节点
        if newState.NextNode != "" {
            currentNodeID = newState.NextNode
        } else {
            // 通过边查找下一个节点
            nextNode := e.findNextNode(currentNodeID, state)
            if nextNode == "" {
                break
            }
            currentNodeID = nextNode
        }
    }

    state.Output = fmt.Sprintf("%v", state.Variables["llm_output"])
    return state, nil
}

func (e *Engine) findNextNode(currentNodeID string, state *State) string {
    for _, edge := range e.Edges {
        if edge.Source == currentNodeID {
            if edge.Condition == nil || edge.Condition(state) {
                return edge.Target
            }
        }
    }
    return ""
}
```

```go
        "output":  state.Output,
        "history": state.History,
    }, "success")
}
```

```go
package services

import (

// LLMProvider LLM 提供商
type LLMProvider string

const (
    ProviderOpenAI    LLMProvider = "openai"
    ProviderAnthropic LLMProvider = "anthropic"
    ProviderAzure     LLMProvider = "azure"
)

// LLMConfig LLM 配置
type LLMConfig struct {
    Provider  LLMProvider
    APIKey    string
    BaseURL   string
    Model     string
    MaxTokens int
}

// LLMManager LLM 管理器
type LLMManager struct {
    configs map[LLMProvider]*LLMConfig
    clients map[LLMProvider]interface{}
    mu      sync.RWMutex
}

func NewLLMManager() *LLMManager {
    return &LLMManager{
        configs: make(map[LLMProvider]*LLMConfig),
        clients: make(map[LLMProvider]interface{}),
    }
}

func (m *LLMManager) Register(provider LLMProvider, config *LLMConfig) {
    m.mu.Lock()
    defer m.mu.Unlock()

    m.configs[provider] = config

    // 根据提供商创建客户端
    switch provider {
    case ProviderOpenAI:
        // 创建 OpenAI 客户端
        m.clients[provider] = nil // 实际实现
    case ProviderAnthropic:
        // 创建 Anthropic 客户端
        m.clients[provider] = nil
    }
}

func (m *LLMManager) Chat(ctx context.Context, provider LLMProvider, messages []Message) (string, error) {
    m.mu.RLock()
    config, exists := m.configs[provider]
    m.mu.RUnlock()

    if !exists {
        return "", fmt.Errorf("提供商 %s 未注册", provider)
    }

    // 根据提供商调用对应的 API
    switch provider {
    case ProviderOpenAI:
        return m.chatOpenAI(ctx, config, messages)
    case ProviderAnthropic:
        return m.chatAnthropic(ctx, config, messages)
    default:
        return "", fmt.Errorf("不支持的提供商: %s", provider)
    }
}

func (m *LLMManager) chatOpenAI(ctx context.Context, config *LLMConfig, messages []Message) (string, error) {
    // 使用 go-openai 库
    return "OpenAI 响应", nil
}

func (m *LLMManager) chatAnthropic(ctx context.Context, config *LLMConfig, messages []Message) (string, error) {
    // 使用 anthropic API
    return "Anthropic 响应", nil
}

func (m *LLMManager) ListProviders() []LLMProvider {
    m.mu.RLock()
    defer m.mu.RUnlock()

    var providers []LLMProvider
    for provider := range m.configs {
        providers = append(providers, provider)
```

### 2.2 多 LLM 路由

```go
// internal/services/router.go
package services

import (
    "context"
    "sync"
)

// LLMRouter LLM 路由器
type LLMRouter struct {
    manager     *LLMManager
    defaultProv LLMProvider
    fallbacks   []LLMProvider
    mu          sync.RWMutex
}

func NewLLMRouter(manager *LLMManager, defaultProvider LLMProvider) *LLMRouter {
    return &LLMRouter{
        manager:     manager,
        defaultProv: defaultProvider,
        fallbacks:   []LLMProvider{},
    }
}

func (r *LLMRouter) AddFallback(provider LLMProvider) {
    r.mu.Lock()
    defer r.mu.Unlock()
    r.fallbacks = append(r.fallbacks, provider)
}

func (r *LLMRouter) Chat(ctx context.Context, messages []Message) (string, error) {
    // 尝试默认提供商
    response, err := r.manager.Chat(ctx, r.defaultProv, messages)
    if err == nil {
        return response, nil
    }

    // 尝试备选提供商
    r.mu.RLock()
    fallbacks := r.fallbacks
    r.mu.RUnlock()

    for _, provider := range fallbacks {
        response, err := r.manager.Chat(ctx, provider, messages)
        if err == nil {
            return response, nil
        }
    }

    return "", fmt.Errorf("所有 LLM 提供商都失败")
}
```

### 4.2 客服对话服务

```go
// internal/services/customer_service.go
package services

import (
    "context"
    "fmt"
    "sync"
    "time"

    "github.com/google/uuid"
)

// Conversation 对话
type Conversation struct {
    ID        string     `json:"id"`
    CustomerID string    `json:"customer_id"`
    AgentID   string     `json:"agent_id,omitempty"`
    Status    string     `json:"status"` // "waiting", "active", "closed"
    Messages  []Message  `json:"messages"`
    CreatedAt time.Time  `json:"created_at"`
    UpdatedAt time.Time  `json:"updated_at"`
}

// Message 消息
type Message struct {
    ID             string    `json:"id"`
    ConversationID string    `json:"conversation_id"`
    Sender         string    `json:"sender"` // "customer", "agent", "system"
    Content        string    `json:"content"`
    MessageType    string    `json:"message_type"` // "text", "image", "file"
    CreatedAt      time.Time `json:"created_at"`
}

// CustomerService 客服服务
type CustomerService struct {
    conversations map[string]*Conversation
    llmRouter     *LLMRouter
    ragService    *RAGService
    mu            sync.RWMutex
}

func NewCustomerService(llmRouter *LLMRouter, ragService *RAGService) *CustomerService {
    return &CustomerService{
        conversations: make(map[string]*Conversation),
        llmRouter:     llmRouter,
        ragService:    ragService,
    }
}

// CreateConversation 创建对话
func (s *CustomerService) CreateConversation(customerID string) *Conversation {
    s.mu.Lock()
    defer s.mu.Unlock()

    conv := &Conversation{
        ID:         uuid.New().String(),
        CustomerID: customerID,
        Status:     "waiting",
        Messages:   []Message{},
        CreatedAt:  time.Now(),
        UpdatedAt:  time.Now(),
    }

    s.conversations[conv.ID] = conv
    return conv
}

// SendMessage 发送消息
func (s *CustomerService) SendMessage(ctx context.Context, conversationID, sender, content string) (*Message, error) {
    s.mu.Lock()
    conv, exists := s.conversations[conversationID]
    if !exists {
        s.mu.Unlock()
        return nil, fmt.Errorf("对话不存在")
    }

    // 添加客户消息
    msg := Message{
        ID:             uuid.New().String(),
        ConversationID: conversationID,
        Sender:         sender,
        Content:        content,
        MessageType:    "text",
        CreatedAt:      time.Now(),
    }

    conv.Messages = append(conv.Messages, msg)
    conv.UpdatedAt = time.Now()
    s.mu.Unlock()

    // 如果是客户消息，生成 AI 回复
    if sender == "customer" {
        go s.generateAIResponse(ctx, conversationID)
    }

    return &msg, nil
}

// generateAIResponse 生成 AI 回复
func (s *CustomerService) generateAIResponse(ctx context.Context, conversationID string) {
    s.mu.RLock()
    conv := s.conversations[conversationID]
    s.mu.RUnlock()

    // 获取最近的消息作为上下文
    var messages []Message
    if len(conv.Messages) > 10 {
        messages = conv.Messages[len(conv.Messages)-10:]
    } else {
        messages = conv.Messages
    }

    // 搜索知识库获取相关信息
    lastMsg := conv.Messages[len(conv.Messages)-1].Content
    ragContext, _ := s.ragService.Query(ctx, lastMsg)

    // 构建 prompt
    prompt := fmt.Sprintf(`你是一个智能客服助手。请根据以下信息回答客户问题。

相关知识：
%s

客户问题：
%s

请用友好、专业的语气回答：`, ragContext, lastMsg)

    // 调用 LLM
    response, err := s.llmRouter.Chat(ctx, []Message{
        {Role: "user", Content: prompt},
    })

    if err != nil {
        response = "抱歉，系统暂时无法处理您的请求，请稍后再试。"
    }

    // 添加 AI 回复
    s.mu.Lock()
    conv = s.conversations[conversationID]
    aiMsg := Message{
        ID:             uuid.New().String(),
        ConversationID: conversationID,
        Sender:         "system",
        Content:        response,
        MessageType:    "text",
        CreatedAt:      time.Now(),
    }
    conv.Messages = append(conv.Messages, aiMsg)
    conv.UpdatedAt = time.Now()
    s.mu.Unlock()
}

// GetConversation 获取对话
func (s *CustomerService) GetConversation(id string) (*Conversation, bool) {
    s.mu.RLock()
    defer s.mu.RUnlock()
    conv, exists := s.conversations[id]
    return conv, exists
}

// ListConversations 列出对话
func (s *CustomerService) ListConversations(status string) []*Conversation {
    s.mu.RLock()
    defer s.mu.RUnlock()

    var result []*Conversation
    for _, conv := range s.conversations {
        if status == "" || conv.Status == status {
            result = append(result, conv)
        }
    }
    return result
}
```

```go
// internal/handlers/customer_service.go
package handlers

import (
    "net/http"

    "github.com/ai-agent-backend/internal/services"
    "github.com/ai-agent-backend/internal/utils"
    "github.com/gin-gonic/gin"
)

var csService *services.CustomerService

func InitCustomerService(llmRouter *services.LLMRouter, ragService *services.RAGService) {
    csService = services.NewCustomerService(llmRouter, ragService)
}

type CreateConversationRequest struct {
    CustomerID string `json:"customer_id" binding:"required"`
}

func CreateConversation(c *gin.Context) {
    var req CreateConversationRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        utils.ErrorResponse(c, "请提供客户 ID", http.StatusBadRequest)
        return
    }

    conv := csService.CreateConversation(req.CustomerID)
    utils.CreatedResponse(c, conv, "对话创建成功")
}

type SendMessageRequest struct {
    Content string `json:"content" binding:"required"`
    Sender  string `json:"sender" binding:"required"`
}

func SendMessage(c *gin.Context) {
    conversationID := c.Param("id")

    var req SendMessageRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        utils.ErrorResponse(c, "请提供消息内容", http.StatusBadRequest)
        return
    }

    msg, err := csService.SendMessage(c.Request.Context(), conversationID, req.Sender, req.Content)
    if err != nil {
        utils.ErrorResponse(c, err.Error(), http.StatusInternalServerError)
        return
    }

    utils.SuccessResponse(c, msg, "success")
}

func GetConversation(c *gin.Context) {
    id := c.Param("id")
    conv, exists := csService.GetConversation(id)
    if !exists {
        utils.ErrorResponse(c, "对话不存在", http.StatusNotFound)
        return
    }
    utils.SuccessResponse(c, conv, "success")
}

func ListConversations(c *gin.Context) {
    status := c.Query("status")
    convs := csService.ListConversations(status)
    utils.SuccessResponse(c, convs, "success")
}
```
