### 1.2 实现缓冲记忆

```go
// internal/services/memory.go
package services

import (
    "sync"
    "time"
)

type MemoryEntry struct {
    Content   string
    Timestamp time.Time
}

type MemoryService struct {
    memories map[string][]MemoryEntry
    mu       sync.RWMutex
}

func NewMemoryService() *MemoryService {
    return &MemoryService{
        memories: make(map[string][]MemoryEntry),
    }
}

// 获取会话记忆
func (s *MemoryService) GetMemory(sessionId string) []MemoryEntry {
    s.mu.RLock()
    defer s.mu.RUnlock()
    return s.memories[sessionId]
}

// 添加记忆
func (s *MemoryService) AddMemory(sessionId string, content string) {
    s.mu.Lock()
    defer s.mu.Unlock()

    memories := s.memories[sessionId]
    memories = append(memories, MemoryEntry{
        Content:   content,
        Timestamp: time.Now(),
    })

    // 保留最近 20 条
    if len(memories) > 20 {
        memories = memories[len(memories)-20:]
    }

    s.memories[sessionId] = memories
}

// 清除记忆
func (s *MemoryService) ClearMemory(sessionId string) {
    s.mu.Lock()
    defer s.mu.Unlock()
    delete(s.memories, sessionId)
}

// 获取历史消息
func (s *MemoryService) GetHistoryMessages(sessionId string) []Message {
    memories := s.GetMemory(sessionId)
    var messages []Message

    for i, entry := range memories {
        role := "assistant"
        if i%2 == 0 {
            role = "user"
        }
        messages = append(messages, Message{
            Role:    role,
            Content: entry.Content,
        })
    }

    return messages
}
```

// internal/services/document_loader.go
package services

import (
    "bufio"
    "os"
    "strings"
)

type Document struct {
    PageContent string
    Metadata    map[string]interface{}
}

type DocumentLoader struct {
    ChunkSize    int
    ChunkOverlap int
}

func NewDocumentLoader(chunkSize, chunkOverlap int) *DocumentLoader {
    return &DocumentLoader{
        ChunkSize:    chunkSize,
        ChunkOverlap: chunkOverlap,
    }
}

// 加载文本文件
func (l *DocumentLoader) LoadTextFile(filePath string) ([]Document, error) {
    file, err := os.Open(filePath)
    if err != nil {
        return nil, err
    }
    defer file.Close()

    var content strings.Builder
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        content.WriteString(scanner.Text())
        content.WriteString("\n")
    }

    docs := []Document{
        {
            PageContent: content.String(),
            Metadata:    map[string]interface{}{"source": filePath},
        },
    }

    return l.SplitDocuments(docs), nil
}

// 加载 Markdown 文件
func (l *DocumentLoader) LoadMarkdownFile(filePath string) ([]Document, error) {
    return l.LoadTextFile(filePath)
}

// 分割文档
func (l *DocumentLoader) SplitDocuments(docs []Document) []Document {
    var result []Document

    for _, doc := range docs {
        content := doc.PageContent
        for i := 0; i < len(content); i += l.ChunkSize - l.ChunkOverlap {
            end := i + l.ChunkSize
            if end > len(content) {
                end = len(content)
            }

            chunk := Document{
                PageContent: content[i:end],
                Metadata:    doc.Metadata,
            }
            result = append(result, chunk)

            if end == len(content) {
                break
            }
        }
    }

    return result
}

// 加载目录
func (l *DocumentLoader) LoadDirectory(dirPath string) ([]Document, error) {
    var allDocs []Document

    entries, err := os.ReadDir(dirPath)
    if err != nil {
        return nil, err
    }

    for _, entry := range entries {
        fullPath := dirPath + "/" + entry.Name()

        if entry.IsDir() {
            subDocs, err := l.LoadDirectory(fullPath)
            if err != nil {
                continue
            }
            allDocs = append(allDocs, subDocs...)
        } else if strings.HasSuffix(entry.Name(), ".txt") {
            docs, err := l.LoadTextFile(fullPath)
            if err == nil {
                allDocs = append(allDocs, docs...)
            }
        } else if strings.HasSuffix(entry.Name(), ".md") {
            docs, err := l.LoadMarkdownFile(fullPath)
            if err == nil {
                allDocs = append(allDocs, docs...)
            }
        }
    }

    return allDocs, nil
}

### 2.2 向量数据库服务

```go
// internal/services/vector_store.go
package services

import (
    "math"
    "sort"
)

type DocumentChunk struct {
    ID         string
    DocumentID string
    Content    string
    Embedding  []float64
}

type VectorStoreService struct {
    chunks []DocumentChunk
}

func NewVectorStoreService() *VectorStoreService {
    return &VectorStoreService{}
}

// 生成简单向量（模拟）
func (s *VectorStoreService) GenerateEmbedding(text string) []float64 {
    dim := 100
    embedding := make([]float64, dim)

    words := strings.Fields(text)
    for _, word := range words {
        hash := 0
        for _, c := range word {
            hash = (hash*31 + int(c)) % dim
        }
        embedding[hash] += 1.0
    }

    // 归一化
    norm := 0.0
    for _, v := range embedding {
        norm += v * v
    }
    norm = math.Sqrt(norm)

    if norm > 0 {
        for i := range embedding {
            embedding[i] /= norm
        }
    }

    return embedding
}

// 存储文档
func (s *VectorStoreService) AddDocuments(docs []Document) {
    for _, doc := range docs {
        chunk := DocumentChunk{
            ID:         generateID(),
            DocumentID: doc.Metadata["id"].(string),
            Content:    doc.PageContent,
            Embedding:  s.GenerateEmbedding(doc.PageContent),
        }
        s.chunks = append(s.chunks, chunk)
    }
}

// 相似度搜索
func (s *VectorStoreService) SimilaritySearch(query string, k int) []SearchResult {
    queryEmbedding := s.GenerateEmbedding(query)

    type scored struct {
        chunk DocumentChunk
        score float64
    }

    var scoredChunks []scored
    for _, chunk := range s.chunks {
        score := cosineSimilarity(queryEmbedding, chunk.Embedding)
        scoredChunks = append(scoredChunks, scored{chunk: chunk, score: score})
    }

    sort.Slice(scoredChunks, func(i, j int) bool {
        return scoredChunks[i].score > scoredChunks[j].score
    })

    if k > len(scoredChunks) {
        k = len(scoredChunks)
    }

    var results []SearchResult
    for i := 0; i < k; i++ {
        results = append(results, SearchResult{
            Chunk: scoredChunks[i].chunk,
            Score: scoredChunks[i].score,
        })
    }

    return results
}

// 余弦相似度
func cosineSimilarity(a, b []float64) float64 {
    dotProduct := 0.0
    normA := 0.0
    normB := 0.0

    for i := range a {
        dotProduct += a[i] * b[i]
        normA += a[i] * a[i]
        normB += b[i] * b[i]
    }

    normA = math.Sqrt(normA)
    normB = math.Sqrt(normB)

    if normA == 0 || normB == 0 {
        return 0
    }

    return dotProduct / (normA * normB)
}

type SearchResult struct {
    Chunk DocumentChunk
    Score float64
}
```

// internal/services/rag.go
package services

import (
    "context"
    "fmt"
    "strings"

    "github.com/sashabaranov/go-openai"
)

type RAGService struct {
    llmClient    *openai.Client
    vectorStore  *VectorStoreService
    docLoader    *DocumentLoader
}

func NewRAGService(apiKey string) *RAGService {
    client := openai.NewClient(apiKey)
    return &RAGService{
        llmClient:   client,
        vectorStore: NewVectorStoreService(),
        docLoader:   NewDocumentLoader(500, 50),
    }
}

// 构建知识库
func (s *RAGService) BuildKnowledgeBase(filePaths []string) (int, error) {
    var allDocs []Document

    for _, filePath := range filePaths {
        info, err := os.Stat(filePath)
        if err != nil {
            continue
        }

        if info.IsDir() {
            docs, err := s.docLoader.LoadDirectory(filePath)
            if err == nil {
                allDocs = append(allDocs, docs...)
            }
        } else {
            docs, err := s.docLoader.LoadTextFile(filePath)
            if err == nil {
                allDocs = append(allDocs, docs...)
            }
        }
    }

    s.vectorStore.AddDocuments(allDocs)
    return len(allDocs), nil
}

// 查询知识库
func (s *RAGService) Query(ctx context.Context, question string) (string, error) {
    // 搜索相关文档
    results := s.vectorStore.SimilaritySearch(question, 3)

    if len(results) == 0 {
        return "未找到相关信息", nil
    }

    // 构建上下文
    var context strings.Builder
    context.WriteString("相关知识：\n")
    for _, result := range results {
        context.WriteString("- ")
        context.WriteString(result.Chunk.Content)
        context.WriteString("\n")
    }

    // 调用 LLM 生成回答
    prompt := fmt.Sprintf(`你是一个专业的问答助手。请根据以下上下文信息回答用户的问题。
如果上下文中没有相关信息，请说明你不确定，不要编造答案。

上下文：
%s

问题：%s

回答：`, context.String(), question)

    resp, err := s.llmClient.CreateChatCompletion(ctx, openai.ChatCompletionRequest{
        Model: openai.GPT3Dot5Turbo,
        Messages: []openai.ChatCompletionMessage{
            {Role: openai.ChatMessageRoleUser, Content: prompt},
        },
    })

    if err != nil {
        return "", fmt.Errorf("LLM 调用失败: %w", err)
    }

    return resp.Choices[0].Message.Content, nil
}

// 带来源的查询
func (s *RAGService) QueryWithSources(ctx context.Context, question string) (map[string]interface{}, error) {
    results := s.vectorStore.SimilaritySearch(question, 3)
    answer, err := s.Query(ctx, question)
    if err != nil {
        return nil, err
    }

    var sources []map[string]interface{}
    for _, result := range results {
        sources = append(sources, map[string]interface{}{
            "content": result.Chunk.Content,
            "score":   result.Score,
        })
    }

    return map[string]interface{}{
        "answer":  answer,
        "sources": sources,
    }, nil
}

// internal/services/function_calling.go
package services

import (
    "context"
    "encoding/json"
    "fmt"

    "github.com/sashabaranov/go-openai"
)

type ToolCall struct {
    Name      string
    Arguments map[string]interface{}
}

type FunctionCallingService struct {
    client *openai.Client
    tools  []openai.Tool
}

func NewFunctionCallingService(apiKey string) *FunctionCallingService {
    client := openai.NewClient(apiKey)
    service := &FunctionCallingService{
        client: client,
    }
    service.defineTools()
    return service
}

func (s *FunctionCallingService) defineTools() {
    s.tools = []openai.Tool{
        {
            Type: "function",
            Function: &openai.FunctionDefinition{
                Name:        "get_weather",
                Description: "获取指定城市的天气信息",
                Parameters: map[string]interface{}{
                    "type": "object",
                    "properties": map[string]interface{}{
                        "city": map[string]interface{}{
                            "type":        "string",
                            "description": "城市名称，如：北京、上海",
                        },
                    },
                    "required": []string{"city"},
                },
            },
        },
        {
            Type: "function",
            Function: &openai.FunctionDefinition{
                Name:        "calculate",
                Description: "计算数学表达式",
                Parameters: map[string]interface{}{
                    "type": "object",
                    "properties": map[string]interface{}{
                        "expression": map[string]interface{}{
                            "type":        "string",
                            "description": "数学表达式",
                        },
                    },
                    "required": []string{"expression"},
                },
            },
        },
    }
}

func (s *FunctionCallingService) ExecuteFunction(name string, args map[string]interface{}) interface{} {
    switch name {
    case "get_weather":
        city := args["city"].(string)
        weatherData := map[string]map[string]string{
            "北京": {"temp": "22°C", "condition": "晴朗"},
            "上海": {"temp": "25°C", "condition": "多云"},
            "广州": {"temp": "28°C", "condition": "小雨"},
        }
        if data, ok := weatherData[city]; ok {
            return data
        }
        return map[string]string{"temp": "未知", "condition": "未知"}
    case "calculate":
        expression := args["expression"].(string)
        // 简单的表达式计算
        return fmt.Sprintf("计算结果: %s", expression)
    default:
        return fmt.Sprintf("未知函数: %s", name)
    }
}

func (s *FunctionCallingService) ChatWithTools(ctx context.Context, messages []openai.ChatCompletionMessage) (string, error) {
    resp, err := s.client.CreateChatCompletion(ctx, openai.ChatCompletionRequest{
        Model:    openai.GPT3Dot5Turbo1106,
        Messages: messages,
        Tools:    s.tools,
    })

    if err != nil {
        return "", err
    }

    responseMessage := resp.Choices[0].Message

    // 如果需要调用函数
    if len(responseMessage.ToolCalls) > 0 {
        // 执行工具调用
        for _, toolCall := range responseMessage.ToolCalls {
            var args map[string]interface{}
            json.Unmarshal([]byte(toolCall.Function.Arguments), &args)

            result := s.ExecuteFunction(toolCall.Function.Name, args)

            // 添加工具结果到消息
            messages = append(messages, responseMessage.ToMessage())
            messages = append(messages, openai.ChatCompletionMessage{
                Role:       openai.ChatMessageRoleTool,
                Content:    fmt.Sprintf("%v", result),
                ToolCallID: toolCall.ID,
            })
        }

        // 再次调用模型
        secondResp, err := s.client.CreateChatCompletion(ctx, openai.ChatCompletionRequest{
            Model:    openai.GPT3Dot5Turbo1106,
            Messages: messages,
            Tools:    s.tools,
        })

        if err != nil {
            return "", err
        }

        return secondResp.Choices[0].Message.Content, nil
    }

    return responseMessage.Content, nil
}

// internal/services/agent.go
package services

import (
    "context"
    "fmt"

    "github.com/sashabaranov/go-openai"

type AgentService struct {
    llmClient *openai.Client
    tools     []openai.Tool
}

func NewAgentService(apiKey string) *AgentService {
    client := openai.NewClient(apiKey)
    return &AgentService{
        llmClient: client,
        tools: []openai.Tool{
            {
                Type: "function",
                Function: &openai.FunctionDefinition{
                    Name:        "search_web",
                    Description: "搜索网络信息",
                    Parameters: map[string]interface{}{
                        "type": "object",
                        "properties": map[string]interface{}{
                            "query": map[string]interface{}{
                                "type":        "string",
                                "description": "搜索查询",
                            },
                        },
                        "required": []string{"query"},
                    },
                },
            },
            {
                Type: "function",
                Function: &openai.FunctionDefinition{
                    Name:        "calculate",
                    Description: "计算数学表达式",
                    Parameters: map[string]interface{}{
                        "type": "object",
                        "properties": map[string]interface{}{
                            "expression": map[string]interface{}{
                                "type":        "string",
                                "description": "数学表达式",
                            },
                        },
                        "required": []string{"expression"},
                    },
                },
            },
        },
    }
}

func (s *AgentService) Chat(ctx context.Context, input string) (string, error) {
    messages := []openai.ChatCompletionMessage{
        {
            Role:    openai.ChatMessageRoleSystem,
            Content: "你是一个有用的助手，可以使用工具来回答问题。",
        },
        {
            Role:    openai.ChatMessageRoleUser,
            Content: input,
        },
    }

    // 最大迭代次数，防止无限循环
    maxIterations := 5
    for i := 0; i < maxIterations; i++ {
        resp, err := s.llmClient.CreateChatCompletion(ctx, openai.ChatCompletionRequest{
            Model:    openai.GPT3Dot5Turbo,
            Messages: messages,
            Tools:    s.tools,
        })

        if err != nil {
            return "", err
        }

        responseMessage := resp.Choices[0].Message

        // 如果没有工具调用，返回结果
        if len(responseMessage.ToolCalls) == 0 {
            return responseMessage.Content, nil
        }

        // 执行工具调用
        messages = append(messages, responseMessage.ToMessage())

        for _, toolCall := range responseMessage.ToolCalls {
            result := s.executeTool(toolCall.Function.Name, toolCall.Function.Arguments)
            messages = append(messages, openai.ChatCompletionMessage{
                Role:       openai.ChatMessageRoleTool,
                Content:    result,
                ToolCallID: toolCall.ID,
            })
        }
    }

    return "达到最大迭代次数", nil
}

func (s *AgentService) executeTool(name string, args string) string {
    switch name {
    case "search_web":
        return "搜索结果：这是一个模拟的搜索结果..."

## 5. LangGraph Go 入门

### 5.1 LangGraph 概念

LangGraph 是一个用于构建有状态、多步骤 Agent 应用的框架。在 Go 中，我们可以使用类似的概念来实现工作流。

### 5.2 简单工作流实现

```go
// internal/services/workflow.go
package services

import (
    "context"
    "fmt"
)

type Node interface {
    Execute(ctx context.Context, state map[string]interface{}) (map[string]interface{}, error)
}

type Edge struct {
    From      string
    To        string
    Condition func(state map[string]interface{}) bool
}

type Workflow struct {
    Name  string
    Nodes map[string]Node
    Edges []Edge
}

func NewWorkflow(name string) *Workflow {
    return &Workflow{
        Name:  name,
        Nodes: make(map[string]Node),
    }
}

func (w *Workflow) AddNode(name string, node Node) {
    w.Nodes[name] = node
}

func (w *Workflow) AddEdge(from, to string, condition func(state map[string]interface{}) bool) {
    w.Edges = append(w.Edges, Edge{
        From:      from,
        To:        to,
        Condition: condition,
    })
}

func (w *Workflow) Execute(ctx context.Context, initialState map[string]interface{}) (map[string]interface{}, error) {
    state := initialState
    currentNode := "start"

    for {
        node, exists := w.Nodes[currentNode]
        if !exists {
            break
        }

        var err error
        state, err = node.Execute(ctx, state)
        if err != nil {
            return nil, fmt.Errorf("节点 %s 执行失败: %w", currentNode, err)
        }

        // 查找下一个节点
        nextNode := ""
        for _, edge := range w.Edges {
            if edge.From == currentNode {
                if edge.Condition == nil || edge.Condition(state) {
                    nextNode = edge.To
                    break
                }
            }
        }

        if nextNode == "" {
            break
        }

        currentNode = nextNode
    }

    return state, nil
}

// 示例节点
type StartNode struct{}

func (n *StartNode) Execute(ctx context.Context, state map[string]interface{}) (map[string]interface{}, error) {
    state["started"] = true
    return state, nil
}

type LLMNode struct {
    Service *LLMService
}

func (n *LLMNode) Execute(ctx context.Context, state map[string]interface{}) (map[string]interface{}, error) {
    input := state["input"].(string)
    // 调用 LLM
    state["llm_output"] = "LLM 处理结果: " + input
    return state, nil
}

type EndNode struct{}

func (n *EndNode) Execute(ctx context.Context, state map[string]interface{}) (map[string]interface{}, error) {
    state["completed"] = true
    return state, nil
}
```

### 5.3 React 聊天界面（带记忆）

```tsx
// src/components/ChatWithMemory.tsx
import React, { useState, useRef, useEffect } from 'react';
import { Input, Button, List, Typography, Card } from 'antd';
import { SendOutlined, DeleteOutlined } from '@ant-design/icons';

const { TextArea } = Input;
const { Text } = Typography;

interface Message {
  role: 'user' | 'assistant';
  content: string;
}

export const ChatWithMemory: React.FC = () => {
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const [sessionId] = useState(() => `session_${Date.now()}`);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages]);

  const handleSend = async () => {
    if (!input.trim() || loading) return;

    const userMessage = input.trim();
    setInput('');
    setMessages((prev) => [...prev, { role: 'user', content: userMessage }]);
    setLoading(true);

    try {
      const response = await fetch('/api/memory/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          session_id: sessionId,
          message: userMessage,
        }),
      });

      const data = await response.json();
      const assistantMessage = data.data.response;
      setMessages((prev) => [...prev, { role: 'assistant', content: assistantMessage }]);
    } catch (error) {
      console.error('发送失败:', error);
      setMessages((prev) => [
        ...prev,
        { role: 'assistant', content: '抱歉，发送消息时出现错误。' },
      ]);
    } finally {
      setLoading(false);
    }
  };

  const handleClear = async () => {
    try {
      await fetch('/api/memory/clear', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ session_id: sessionId }),
      });
      setMessages([]);
    } catch (error) {
      console.error('清除失败:', error);
    }
  };

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100vh', maxWidth: 800, margin: '0 auto' }}>
      <div style={{ padding: 16, borderBottom: '1px solid #f0f0f0', display: 'flex', justifyContent: 'space-between' }}>
        <Text strong>带记忆的聊天</Text>
        <Button icon={<DeleteOutlined />} onClick={handleClear}>
          清除记忆
        </Button>
      </div>

      <div style={{ flex: 1, overflowY: 'auto', padding: 16 }}>
        {messages.map((message, index) => (
          <div
            key={index}
            style={{
              display: 'flex',
              justifyContent: message.role === 'user' ? 'flex-end' : 'flex-start',
              marginBottom: 16,
            }}
          >
            <Card
              size="small"
              style={{
                maxWidth: '70%',
                background: message.role === 'user' ? '#1677ff' : '#f5f5f5',
              }}
            >
              <Text style={{ color: message.role === 'user' ? '#fff' : '#333' }}>
                {message.content}
              </Text>
            </Card>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      <div style={{ padding: 16, borderTop: '1px solid #f0f0f0' }}>
        <div style={{ display: 'flex', gap: 8 }}>
          <TextArea
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onPressEnter={(e) => {
              if (!e.shiftKey) {
                e.preventDefault();
                handleSend();
              }
            }}
            placeholder="输入消息..."
            autoSize={{ minRows: 1, maxRows: 4 }}
            disabled={loading}
            style={{ flex: 1 }}
          />
          <Button
            type="primary"
            icon={<SendOutlined />}
            onClick={handleSend}
            loading={loading}
          >
            发送
          </Button>
        </div>
      </div>
    </div>
  );
};
```

## 📝 常见问题

### Q1: RAG 检索结果不相关

**解决方案：**
1. 调整 ChunkSize 和 ChunkOverlap
2. 使用更好的 Embedding 模型
3. 实现混合检索（向量 + 关键词）
4. 添加重排序（ReRank）

### Q2: Agent 陷入无限循环

**解决方案：**
1. 设置 maxIterations 限制
2. 改进 Prompt，明确终止条件
3. 添加循环检测逻辑

### Q3: 向量数据库内存不足

**解决方案：**
1. 使用磁盘持久化存储
2. 考虑使用 Milvus 等分布式向量数据库
3. 减少文档数量或使用摘要
